\begin{figure}[htbp]
    \centering
    \tikzfig{cha-vt-diagram-main.tex}
    \caption{The $v$--$t$ diagram for constant acceleration. Motion with a constant acceleration $a$ corresponds to a straight line with slope $a$. The line starts at the initial velocity $v_0$ and climbs at the rate $a$, so over a time interval $\Delta t$ it gains $a\Delta t$ and reaches $v_f = v_0 + a \Delta t$. The area under it is the displacement $\Delta x$, and it's the sum of a rectangle $v_0 \Delta t$ and a triangle $\frac{1}{2}(a \Delta t)(\Delta t)$ sitting on top.}
    \label{fig:vt-diagram-main}
  \end{figure}
