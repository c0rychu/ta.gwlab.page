%% Header for the standalone per-chapter PDFs.
%%
%% A chapter PDF leaves here on its own -- downloaded, printed, mailed around --
%% with no title page in front of it, so it has to say who wrote it, where the
%% current and complete version lives, and how old the copy in hand is. This
%% file is where that text is edited; gwbook.cls only provides the hook.
%%
%% \input from the preamble of book.tex, so it is read by the whole-book build
%% and by every chapter build alike. \ifSubfilesClassLoaded is what separates
%% the two: it is true only when a chapters/*.tex file is being compiled as its
%% own document, which is exactly when the banner is wanted. In the book the
%% title page already carries all three facts, and the hook stays empty.

\ifSubfilesClassLoaded{%
  \chapterbanner{%
    \begingroup
    % The document's own text face, not the sans: this is the first thing on
    % the page and a second typeface up there reads as a sticker slapped on
    % top rather than as part of the notes.
    \normalfont\footnotesize\color{black!60}%
    % \parbox, not a bare paragraph: the banner is typeset inside the chapter
    % title's label box, which is set \huge\bfseries and centred on its own
    % terms. The box gives it back an ordinary \textwidth measure to break in.
    \parbox{\textwidth}{%
      \raggedright
      Made by Cory Chu (UWM) \hfill Built \today\\
      Up-to-date version and complete notes:
      \href{https://ta.gwlab.page/}{https://ta.gwlab.page}%
    }\par
    \vspace{5pt}\hrule height 0.4pt
    \endgroup
    \vspace{26pt}%
  }%
}{}
